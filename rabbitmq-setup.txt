helm repo add bitnami https://charts.bitnami.com/bitnami
helm repo update

kubectl create namespace messaging

vi rabbitmq-values.yaml
auth:
  username: admin
  password: "changeme-strong-password"
  erlangCookie: "changeme-erlang-cookie"

replicaCount: 1

persistence:
  enabled: true
  storageClass: ""
  size: 4Gi

extraPlugins: "rabbitmq_management"

resources:
  requests:
    cpu: 200m
    memory: 256Mi
  limits:
    cpu: 500m
    memory: 512Mi

service:
  type: NodePort

helm install rabbitmq bitnami/rabbitmq -n messaging -f rabbitmq-values.yaml

vi rabbitmq-secret.yaml

apiVersion: v1
kind: Secret
metadata:
  name: rabbitmq-creds
  namespace: microservices
type: Opaque
stringData:
  RABBITMQ_URL: "amqp://admin:changeme-strong-password@rabbitmq.messaging.svc.cluster.local:5672/"

vi audio-service.yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: audio-service
  namespace: microservices
spec:
  replicas: 1
  selector:
    matchLabels:
      app: audio-service
  template:
    metadata:
      labels:
        app: audio-service
    spec:
      containers:
        - name: audio-service
          image: manoharshetty507/audio-converter-service:latest
          imagePullPolicy: Always
          ports:
            - containerPort: 5003
          env:
            - name: RABBITMQ_URL
              valueFrom:
                secretKeyRef:
                  name: rabbitmq-creds
                  key: RABBITMQ_URL
---
apiVersion: v1
kind: Service
metadata:
  name: audio-service
  namespace: microservices
spec:
  selector:
    app: audio-service
  ports:
    - port: 5003
      targetPort: 5003

audio-worker.yaml

apiVersion: apps/v1
kind: Deployment
metadata:
  name: audio-worker
  namespace: microservices
spec:
  replicas: 2
  selector:
    matchLabels:
      app: audio-worker
  template:
    metadata:
      labels:
        app: audio-worker
    spec:
      containers:
        - name: audio-worker
          image: manoharshetty507/audio-worker:latest
          imagePullPolicy: Always
          env:
            - name: RABBITMQ_URL
              valueFrom:
                secretKeyRef:
                  name: rabbitmq-creds
                  key: RABBITMQ_URL
            - name: PYTHONUNBUFFERED
              value: "1"
          resources:
            requests:
              cpu: 100m
              memory: 128Mi
            limits:
              cpu: 300m
              memory: 256Mi

kubectl apply -f audio-service-deployment.yaml
kubectl apply -f audio-worker-deployment.yaml

kubectl get pods -n microservices
kubectl get pods -n messaging

kubectl logs -f deployment/audio-worker -n microservices

kubectl port-forward -n microservices svc/audio-service 5003:5003
